#Text-generation model
import openai
from google.colab import userdata

# Retrieve Gemini API key from Colab secrets
GEMINI_API_KEY = userdata.get('gemni-api-key')

# Initialize the OpenAI client pointing to the Google Gemini API endpoint
client = openai.OpenAI(
    api_key=GEMINI_API_KEY,
    base_url="https://generativelanguage.googleapis.com/v1beta/"
)

# Use the chat completion API with the Gemini 3 Flash Preview model
response = client.chat.completions.create(
    model="gemini-3-flash-preview",
    messages=[
        {"role": "user", "content": "What is the capital of France?"},
        {"role": "assistant", "content": "The capital of France is Paris."},
        

    ],
    temperature=0.7,
)

# Print the result
print(response.choices[0].message.content)

**RUN CELL

** You can continue chat by adding {"role": "user", "content": "Input question"},
